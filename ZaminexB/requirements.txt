# ---------------------------------------------------------------------------

# --- Core framework --------------------------------------------------------
Django==5.2.16
djangorestframework==3.17.1

# --- Database --------------------------------------------------------------
# PostgreSQL driver. `psycopg2-binary` ships prebuilt wheels, so no local
# compiler or libpq headers are required.
psycopg2-binary==2.9.12

# Parses the DATABASE_URL environment variable into Django's DATABASES dict.
dj-database-url==3.1.2

# --- Configuration ---------------------------------------------------------

# --- Media -----------------------------------------------------------------
# Required by Django's ImageField (property gallery, consultant avatars).
pillow==12.3.0

# --- Persian calendar ------------------------------------------------------
# Jalali (Shamsi) date conversion for the Persian monthly revenue chart.
jdatetime==6.1.0

# --- PDF report ------------------------------------------------------------
# Server-side «گزارش کامل» PDF: reportlab renders the document, arabic-reshaper
# joins the Persian letters and python-bidi flips the text to visual order.
reportlab==5.0.1
arabic-reshaper==3.0.1
python-bidi==0.6.11

# --- Search ----------------------------------------------------------------
# Uses PostgreSQL pg_trgm (TrigramSimilarity) and Django postgres search
# for native, field-by-field fuzzy matching with Persian normalization.

# --- WSGI server -----------------------------------------------------------
# `manage.py runserver` is not a production server: it is single-threaded,
# has no process supervision, and says so on every start. Nothing in this
# file previously provided an alternative, so a deployment had no documented
# way to serve the app at all.
#
# Gunicorn serves `config.wsgi:application` — the same entry point that also
# installs the static file handler (see apps/common/staticfiles.py), which is
# what makes `/static/` work with DEBUG off:
#
#   gunicorn config.wsgi:application --bind 0.0.0.0:8000 --workers 4
gunicorn==26.2.0

# --- Cache (Phase 2) -------------------------------------------------------
# Redis-backed cache when REDIS_URL is set; the app falls back to in-process
# LocMem without it (Redis is an optimisation, never a dependency —
# IGNORE_EXCEPTIONS makes a dead Redis a cache miss, not a 500).
django-redis==7.0.0
redis==8.1.0
